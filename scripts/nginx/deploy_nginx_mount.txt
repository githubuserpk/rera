export GCP_REGION=us-central1
export GCP_PROJECT=ai-pk-aiproject

AR_REPO='nginx-mount-repo'
gcloud artifacts repositories create "$AR_REPO" --location="$GCP_REGION" --repository-format=Docker


SERVICE_NAME='nginx-mount'
gcloud builds submit --tag "$GCP_REGION-docker.pkg.dev/$GCP_PROJECT/$AR_REPO/$SERVICE_NAME"


gcloud run deploy "$SERVICE_NAME" \
  --port=8080 \
  --image="$GCP_REGION-docker.pkg.dev/$GCP_PROJECT/$AR_REPO/$SERVICE_NAME" \
  --platform managed \
  --allow-unauthenticated \
  --region=$GCP_REGION \
  --project=$GCP_PROJECT
  --set-env-vars NGINX_CONF=/etc/nginx/external-conf/nginx.conf \
  --volume-mount mount-path=/etc/nginx/external-conf,volume=nginx-config


# create bucket and copy the nginx.conf to it 
gsutil mb gs://bkt-aiguard
gsutil cp nginx.conf gs://bkt-aiguard/nginx.conf


# configure the volume to use GCS
gcloud run services update nginx-service \
  --set-volumes=nginx-config=gs://bkt-aigurard/nginx.conf
